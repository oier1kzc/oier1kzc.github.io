# [B. Matrix of Differences](https://codeforces.com/contest/1783/problem/B)

## Statement

给定正整数 $n \ (2 \leq n \leq 50)$。

对于一个 $n \times n$ 且所有元素都在 ${1, 2, \cdots, n \times n}$ 中的矩阵 $A$，定义其可爱度为最大的 $k$，满足 $1, 2, \cdots, k$ 都在其所有相邻数对的差的绝对值中出现过。

构造一个矩阵，使得可爱度最大。

## Solution

不用矩阵！！！直接序列就可以！！！被骗了一场！！！

设 $m = n \times n$，构造 $[1, m, 2, m - 1, 3, m - 2\cdots]$，相邻数对差绝对值正好遍历 $1, \cdots, m - 1$，卡到上界。

把这个构造蛇形或螺旋填入矩阵中即可，有很多种填入方式。

## Code

```cpp
#include <stdio.h>
#include <string.h>

#include <vector>
#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

#define eb emplace_back

using namespace std;

typedef long long LL;

const int N = 55;

int n, v[N * N], a[N][N];

int main() {
  int task;
  for (scanf("%d", &task); task--; ) {
    scanf("%d", &n);
    int l = 1, r = n * n, m = 0;
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= n; ++j) {
        v[m++] = (m & 1 ? l++ : r--);
      }
    }
    memset(a, 0, sizeof a);
    for (int x = 1, y = 1, i = 0; i < n * n; ) {
      while (y <= n && !a[x][y]) {
        a[x][y++] = v[i++];
      }
      --y, ++x;
      while (x <= n && !a[x][y]) {
        a[x++][y] = v[i++];
      }
      --y, --x;
      while (y && !a[x][y]) {
        a[x][y--] = v[i++];
      }
      ++y, --x;
      while (x && !a[x][y]) {
        a[x--][y] = v[i++];
      }
      ++y, ++x;
    }
    for (int i = 1; i <= n; ++i) {
      for (int j = 1; j <= n; ++j) {
        printf("%d ", a[i][j]);
      }
      puts("");
    }
  }
  return 0;
}
```

# [D. Different Arrays](https://codeforces.com/contest/1783/problem/D)

## Statement

给定正整数 $n \ (3 \leq n \leq 300)$ 和一个长为 $n$ 的非负整数数组 $a _ 1, \cdots, a _ n (0 \leq a _ i \leq 300)$。

做 $n - 2$ 次操作，第 $i$ 次操作，有以下两种选择：

- 将 $a _ i$ 加上 $a _ {i + 1}$，将 $a _ {i + 2}$ 减去 $a _ {i + 1}$。
- 将 $a _ {i + 2}$ 加上 $a _ {i + 1}$，将 $a _ i$ 减去 $a _ {i + 1}$。

求最后可能得到的不同的数组数量，答案对 $998244353$ 取模。

## Solution

设 $s _ i = \sum \limits _ {j = 1} ^ i a _ j$ 是 $a$ 的前缀和。第 $i$ 次操作变为如下两种情况之一：

- 将 $(s _ i, s _ {i + 1})$ 变为 $(2 s _ i - s _ {i + 1}, s _ i)$。
- 将 $(s _ i, s _ {i + 1})$ 变为 $(s _ {i + 1}, 2 s _ {i + 1} - s _ i)$。

于是可以 DP，设 $f _ {i, j}$ 表示做完了前 $i - 1$ 次操作，得到的 $s _ i = j$ 的方案数。转移如下

- $f _ {i, j} = \begin {cases} f _ {i - 1, j} + f _ {i - 1, 2 s _ i - j} & j \neq s _ i \\ f _ {i - 1, j} & j = s _ i\end {cases}$

时间复杂度为 $\mathcal O \left( n \sum \limits _ {i = 1} ^ n a _ i \right)$。

## Code

```cpp
#include <stdio.h>
#include <string.h>

#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

using namespace std;

typedef long long LL;

const int N = 305, M = 2 * N * N;
const int mod = 998244353;

int n, a[N];
int f[M], g[M];

void Add(int &x, int y) {
  if ((x += y) >= mod) {
    x -= mod;
  }
}

int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", a + i);
    a[i] += a[i - 1];
  }
  f[a[1]] = 1;
  for (int i = 2; i < n; ++i) {
    for (int x = 0; x <= 2 * a[i]; ++x) {
      Add(g[x], f[x]);
      if (x != a[i]) {
        Add(g[2 * a[i] - x], f[x]);
      }
    }
    memcpy(f, g, sizeof f);
    memset(g, 0, sizeof g);
  }
  int res = 0;
  for (int x = 0; x < M; ++x) {
    Add(res, f[x]);
  }
  printf("%d\n", res);
  return 0;
}
```

# [E. Game of the Year](https://codeforces.com/contest/1783/problem/E)

## Statement

给定正整数 $n \ (1 \leq n \leq 2 \cdot 10 ^ 5)$。给定长为 $n$ 的数组 $a _ 1, \cdots, a _ n$ 和 $b _ 1, \cdots, b _ n \ (1 \leq a _ i, b _ i \leq n)$。

求所有的 $k \in \{1, 2, \cdots, n\}$，使得对于所有 $i = 1, 2, \cdots, n$，有 $\left \lceil \dfrac {a _ i} {k} \right \rceil \leq \left \lceil \dfrac {b _ i} {k} \right \rceil$。

## Solution

考虑一个 $1, 2, \cdots$ 的数列，将其中前 $k$ 个数分成第一段，再 $k$ 个数分成第二段，以此类推。

则 $\left \lceil \dfrac x k \right \rceil$ 即 $x$ 所在块编号。

如果 $a _ i \leq b _ i$，那么它对所有 $k$ 一定有 $\left \lceil \dfrac {a _ i} k \right \rceil \leq \left \lceil \dfrac {b _ i} k \right \rceil$，可以不用管。

否则一定有 $\left \lceil \dfrac {a _ i} k \right \rceil \geq \left \lceil \dfrac {b _ i} k \right \rceil$，只需要看是否有 $\left \lceil \dfrac {a _ i} k \right \rceil = \left \lceil \dfrac {b _ i} k \right \rceil$，即是否有 $a _ i$ 和 $b _ i$ 在一个块中。

如果在 $[b _ i, a _ i)$ 中存在一个 $k$ 的倍数，也就是一个分割点，则 $a _ i$ 和 $b _ i$ 不在一个块中。

对于所有 $k$，这样的分割点一共是 $\mathcal O(n \log n)$ 个，是可枚举的。

枚举分割点，需要判断是否有 $[b _ i, a _ i)$ 这样的区间跨过分割点，很多种判断方法。代码种是维护一个长为 $n$ 的数组，在 $[a _ i, b _ i)$ 上标为 $1$，判断分割点位置是否为 $0$。也可以直接在 $[a _ i, b _ i)$ 上 $+1$。

## Code

```cpp
#include <stdio.h>
#include <string.h>

#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

using namespace std;

typedef long long LL;

const int N = 200005;
const int mod = 998244353;

int n, a[N], b[N];
int nxt[N];
int res[N], sz;
bool ws[N];

void cmax(int &x, int y) {
  if (x < y) {
    x = y;
  }
}

int main() {
  int task;
  for (scanf("%d", &task); task--; ) {
    scanf("%d", &n);
    memset(nxt + 1, 0, n << 2);
    for (int i = 0; i < n; ++i) {
      scanf("%d", a + i);
    }
    for (int i = 0; i < n; ++i) {
      scanf("%d", b + i);
    }
    for (int i = 0; i < n; ++i) {
      if (b[i] < a[i]) {
        cmax(nxt[b[i]], a[i]);
      }
    }
    memset(ws + 1, 0, n);
    for (int i = 1, j = 1; i <= n; ++i) {
      if (j < i) {
        j = i;
      }
      while (j < nxt[i]) {
        ws[j++] = true;
      }
    }
    sz = 0;
    for (int k = 1; k <= n; ++k) {
      bool flag = true;
      for (int x = k; x <= n; x += k) {
        if (ws[x]) {
          flag = false;
          break;
        }
      }
      if (flag) {
        res[sz++] = k;
      }
    }
    printf("%d\n", sz);
    for (int i = 0; i < sz; ++i) {
      printf("%d ", res[i]);
    }
    puts("");
  }
  return 0;
}
```

# [F. Double Sort II](https://codeforces.com/contest/1783/problem/F)

## Statement

给定正整数 $n \ (2 \leq n \leq 3000)$，给定两个 $1, 2, \cdots, n$ 的排列 $a, b$。

执行如下操作若干次

- 在 $1, 2, \cdots, n$ 中选择一个数 $i$；
- 找到 $x$ 使得 $a _ x = i$，然后交换 $a _ i$ 和 $a _ x$；
- 找到 $y$ 使得 $b _ y = i$，然后交换 $b _ i$ 和 $b _ y$。

目标是将 $a$ 和 $b$ 升序排序。求最少操作次数，并构造操作序列。

可以证明答案总存在，且操作次数不超过 $2n$。

## Solution

事实上，对 $i$ 操作之后，有 $a _ i = b _ i = i$，所以对 $i = 1, 2, \cdots , n$ 都操作一次，可以得到操作次数不超过 $n$。

根据这样的考虑，发现每个 $i$ 最多只被操作一次，所以只需要考虑选哪些点操作，哪些点不操作。

考虑哪些位置不选，要最大化这样的位置数。

在 $a$ 和 $b$ 的置换环上考虑，每个点恰被两个置换环覆盖。

发现每次对于一个点 $u$ 操作，是从 $u$ 所在的两个环上，把 $u$ 扯下来，单独连两个自环上去，这可以看成是将 $u$ 删除。

对于一个点 $u$，设 $a$ 中包含 $u$ 的环是 $p _ 1, p _ 2, \cdots, p _ k$，其中 $p _ 1 = u$，$b$ 中包含 $u$ 的环是 $q _ 1, q _ 2, \cdots, q _ k$，其中 $q _ 1 = u$。如果 $u$ 不选，也就是没被删除，因为最后是 $u$ 上连两个自环，所以 $p _ 2, \cdots, p _ k$ 和 $q _ 2, \cdots, q _ k$ 就一定都要被删除，也就是都要选。

于是可以得到，对于 $a$ 中的置换环 $A$，$b$ 中的置换环 $B$，若 $A \cap B \neq \varnothing$，则 $A, B$ 中总共至多一个点不选。

这也是充分条件。可以构造说明之：任取 $u \in A \cup B$ 做不选的点即可。

将 $a, b$ 中每个置换环都看成点，如果两个置换环有交，则在对应点之间连边，之后每个不选的点可以看成一个匹配，要最大化不选点数，计算最大匹配即可。

这里用匈牙利算法求最大匹配，时间复杂度为 $\mathcal O (n ^ 2)$。

## Code

```cpp
#include <stdio.h>
#include <string.h>

#include <vector>
#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

#define eb emplace_back

using namespace std;

typedef long long LL;

const int N = 3005;
const int mod = 998244353;

int n, a[N], b[N], ida[N], idb[N];
int res[N], sz;
vector<int> e[N];
bool ws[N];
int mat[N];

void prwk(int *a, int *id) {
  int ids = 0;
  for (int x = 1; x <= n; ++x) {
    int t = x;
    ++ids;
    do {
      id[t] = ids;
      t = a[t];
    } while (t != x);
  }
}

bool dfs(int x) {
  ws[x] = true;
  for (int y : e[x]) {
    int &t = mat[y];
    if (!t || !ws[t] && dfs(t)) {
      t = x;
      return true;
    }
  }
  return false;
}

int main() {
  scanf("%d", &n);
  for (int i = 1; i <= n; ++i) {
    scanf("%d", a + i);
  }
  for (int i = 1; i <= n; ++i) {
    scanf("%d", b + i);
  }
  prwk(a, ida);
  prwk(b, idb);
  for (int x = 1; x <= n; ++x) {
    e[ida[x]].eb(idb[x]);
  }
  for (int x = 1; x <= n; ++x) {
    dfs(x);
    memset(ws + 1, 0, n);
  }
  for (int x = 1; x <= n; ++x) {
    int u = ida[x], v = idb[x];
    if (mat[v] == u) {
      mat[v] = 0;
      ws[x] = true;
    }
  }
  for (int x = 1; x <= n; ++x) {
    if (!ws[x]) {
      res[sz++] = x;
    }
  }
  printf("%d\n", sz);
  for (int i = 0; i < sz; ++i) {
    printf("%d ", res[i]);
  }
  puts("");
  return 0;
}
```

# [G. Weighed Tree Radius](https://codeforces.com/contest/1783/problem/G)

## Statement

给定 $n \ (2 \leq n \leq 2 \cdot 10 ^ 5)$ 个点的树，点 $u$ 的初始点权为 $a _ u \ (0 \leq a _ u \leq 10 ^ 6)$。

定义 $d(u, v)$ 表示树上 $u$ 到 $v$ 的简单路径的边数。

定义 $w(u, v) = d(u, v) + a _ v$。

定义 $e(u) = \max \limits _ {1 \leq v \leq n} w(u, v)$。

定义 $r = \min \limits _ {1 \leq u \leq n} e(u)$。

给定 $m (1 \leq m \leq 10 ^ 5)$ 次操作，每次操作给定 $x, y \ (1 \leq x \leq n, 0 \leq y \leq 10 ^ 6)$，表示将 $a _ x$ 修改为 $y$。

每次操作后，求出 $r$。

## Solution

$e(u) = \max \limits _ {1 \leq v \leq n} \{d(u, v) + a _ v\}$ 难以维护。想办法转化。

如果将 $a _ u$ 看成是一条挂在点 $u$ 上的，长为 $a _ u$ 的链，则 $e(u)$ 的值是 $u$ 所能到的最远距离。

但 $\min \limits _ {1 \leq u \leq n} e (u)$ 的值是只能在原树上的点上取的，为了调整这一点，可以在 $u$ 上挂两条长为 $a _ u$ 的链。

这将「带权的」半径完全转为了一般的半径。

考虑求直径，除以 $2$ 上取整即半径。

### SubSolution1

对于树上的一个点集 $S$，定义其直径为 $u$ 到 $v$ 的路径，满足 $u, v \in S$，且 $\forall x, y \in S, d(u, v) \geq d(x, y)$。

有结论：考虑一棵树上的两个点集，它们的并的直径端点，来源于这两棵虚树的 $4$ 个直径端点中之 $2$。

证明：

设原来两个点集为 $A, B$，原来两直径的端点分别为 $x _ A, y _ A$ 和 $x _ B, y _ B$，新的直径端点为 $u, v$。

记 $R _ U (u, v)$ 表示 $u$ 到 $v$ 的路径的点集，设 $R(u, v) = R _ U (u, v) \cup (A \cap B)$，即路径在 $A \cap B$ 中的那部分点。

两条路径的交，若不为空，则仍是一条路径。

设路径 $P = R(u, v) \cup R(x _ A, y _ A)$，$Q = R(u, v) \cup R(x _ B, y _ B)$。

若 $P$ 为空或者 $Q$ 为空，则 $(u, v)$ 均在 $A$ 或 $B$ 中，取原来的直径不劣。

否则，设 $P$ 的端点为 $x _ P, y _ P$，$Q$ 的端点为 $x _ Q, y _ Q$。

看图！

![chain](/images/posts/cf1783-chain.png)

根据对称性，不妨设

- $u \to v$ 的路径为 $u \to x _ P \to y _ P \to x _ Q \to y _ Q \to v$；
- $x _ A \to y _ A$ 的路径为 $x _ A \to x _ P \to y _ P \to y _ A$；
- $x _ B \to y _ B$ 的路径为 $x _ B \to x _ Q \to y _ Q \to y _ B$。

考虑 $\{x _ A, y _ A\}$ 中距离 $y _ P$ 更远的点 $z _ A$ 和 $\{x _ B, y _ B\}$ 中距离 $x _ Q$ 更远的点 $z _ B$，则 $(z _ A, z _ B)$ 的路径长度不小于 $(u, v)$，于是取 $(z _ A, z _ B)$ 一定不劣。

这个结论的意义是，我们可以不需要考虑顺序地，将直径信息合并。具体来说，使用线段树，对每个区间 $[l, r]$，维护以 $[l, r]$ 为点集的直径 $(u, v)$，每个区间可以由两个子区间合并，得到整个区间信息。

每次合并需要 $\mathcal O(1)$ 次查询两点距离，时间复杂度 $\mathcal O((n + m) \text {dis}(n) \log n)$，其中 $\text{dis}(n)$ 为一次查询大小为 $n$ 的树上两点距离的复杂度。

### Code

```cpp
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#include <vector>
#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

#define eb emplace_back

using namespace std;

const int N = 200005, TC = N << 2;

int n, m, a[N];
vector<int> e[N];
struct Data {
  int x, y;
} tr[TC];
int de[N], fa[N], bg[N], ids;
int f[18][N], wg[N];

void dfs1(int x) {
  f[0][ids] = fa[x], bg[x] = ++ids;
  for (int y : e[x]) {
    if (y == fa[x]) {
      continue;
    }
    fa[y] = x, de[y] = de[x] + 1, dfs1(y);
  }
}

int sha(int x, int y) {
  return de[x] < de[y] ? x : y;
}

int dis(int x, int y) {
  if (x == y) {
    return 2 * a[x];
  }
  int ret = a[x] + a[y] + de[x] + de[y];
  x = bg[x], y = bg[y];
  if (x > y) {
    swap(x, y);
  }
  --y;
  int k = wg[y - x + 1];
  int z = sha(f[k][x], f[k][y - (1 << k) + 1]);
  return ret - 2 * de[z];
}

void merge(Data &c, const Data &a, const Data &b) {
  const int (&v)[4] = {a.x, a.y, b.x, b.y};
  int d = -1, t;
  for (int i = 1; i < 4; ++i) {
    for (int j = 0; j < i; ++j) {
      t = dis(v[i], v[j]);
      if (t > d) {
        c.x = v[i], c.y = v[j];
        d = t;
      }
    }
  }
}

#define L (z << 1)
#define R (z << 1 | 1)

void build(int l = 1, int r = n, int z = 1) {
  if (l == r) {
    tr[z].x = tr[z].y = r;
  } else {
    int mid = l + r >> 1;
    build(l, mid, L), build(mid + 1, r, R);
    merge(tr[z], tr[L], tr[R]);
  }
}

void change(int x, int l = 1, int r = n, int z = 1) {
  if (l != r) {
    int mid = l + r >> 1;
    if (x <= mid) {
      change(x, l, mid, L);
    } else {
      change(x, mid + 1, r, R);
    }
    merge(tr[z], tr[L], tr[R]);
  }
}

int main() {
  scanf("%d", &n);
  for (int x = 1; x <= n; ++x) {
    scanf("%d", a + x);
  }
  for (int i = 1; i < n; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    e[x].eb(y), e[y].eb(x);
  }
  dfs1(1);
  for (int x = 2; x <= n; ++x) {
    wg[x] = wg[x >> 1] + 1;
  }
  for (int k = 0; k < 17; ++k) {
    for (int i = 1; i + (1 << k + 1) - 1 <= n; ++i) {
      f[k + 1][i] = sha(f[k][i], f[k][i + (1 << k)]);
    }
  }
  build();
  int m;
  scanf("%d", &m);
  while (m--) {
    int x, y;
    scanf("%d%d", &x, &y);
    a[x] = y, change(x);
    printf("%d\n", dis(tr[1].x, tr[1].y) + 1 >> 1);
  }
  return 0;
}
```

### SubSolution2

考虑一个使 $a _ u$ 增加的操作，它会对所有一端为 $u$（挂在 $u$ 上的链的端点）的路径长，产生一个增量。于是新的直径要么不变，要么就是从 $u$ 出发的一条路径。

如果是后者，根据直径性质，若原来的直径是 $(x, y)$，则新的直径是 $(u, x)$ 或 $(u, y)$。

只需在这三种情况中枚举即可，需要查 $\mathcal O(1)$ 次两点距离。

但不只有使 $a _ u$ 增加的操作，这可以通过线段树分治解决之。具体地，对于所有 $u$，考虑 $a _ u$ 的变化过程，将其按照时间拆成若干区间，放到线段树上做。每次只需要将一个 $a _ u$ 从 $0$ 变为一个非负数，是不减的。

时间复杂度 $\mathcal O((n + m) \text{dis} (n) \log m)$，其中 $\text{dis}(n)$ 为一次查询大小为 $n$ 的树上两点距离的复杂度。

### Code

```cpp
#include <stdio.h>
#include <string.h>

#include <vector>
#include <algorithm>

#define LOG(FMT...) fprintf(stderr, FMT)

#define eb emplace_back

using namespace std;

typedef long long LL;
const int N = 200005, TC = N * 4;

int n, m, ts[N], a[N];
vector<int> e[N];
struct Req {
  int x, c;
};
vector<Req> rqs[TC];
int res[N];
int de[N], fa[N], bg[N], ids;
int wg[N], f[18][N];

int sha(int x, int y) {
  return de[x] < de[y] ? x : y;
}

void dfs(int x) {
  f[0][ids++] = fa[x], bg[x] = ids;
  if (fa[x]) {
    e[x].erase(find(e[x].begin(), e[x].end(), fa[x]));
  }
  for (int y : e[x]) {
    fa[y] = x, de[y] = de[x] + 1, dfs(y);
  }
}

int dis(int x, int y) {
  if (x == y) {
    return 2 * a[x];
  }
  int ret = de[x] + de[y] + a[x] + a[y];
  x = bg[x], y = bg[y];
  if (x > y) {
    swap(x, y);
  }
  --y;
  int k = wg[y - x + 1];
  return ret - 2 * de[sha(f[k][x], f[k][y - (1 << k) + 1])];
}

#define L (z << 1)
#define R (z << 1 | 1)

void setrq(int ql, int qr, int x, int c, int l = 0, int r = m, int z = 1) {
  if (l >= ql && r <= qr) {
    rqs[z].eb((Req){x, c});
  } else {
    int mid = l + r >> 1;
    if (ql <= mid) {
      setrq(ql, qr, x, c, l, mid, L);
    }
    if (mid < qr) {
      setrq(ql, qr, x, c, mid + 1, r, R);
    }
  }
}

struct Diam {
  int x, y, d;
  void upd(int z) {
    int cx = x, cy = y;
    int zz = dis(z, z);
    if (zz > d) {
      cx = cy = z, d = zz;
    }
    if (z == x || z == y) {
      int xy = dis(x, y);
      if (xy > d) {
        cx = cy = z, d = xy;
      }
    } else {
      int xz = dis(x, z), yz = dis(y, z);
      if (xz > d) {
        cx = x, cy = z, d = xz;
      }
      if (yz > d) {
        cx = y, cy = z, d = yz;
      }
    }
    x = cx, y = cy;
  }
} diam;

void solve(int l = 0, int r = m, int z = 1) {
  Diam backup = diam;
  for (const Req &t : rqs[z]) {
    a[t.x] = t.c;
    diam.upd(t.x);
  }
  if (l == r) {
    res[r] = diam.d;
  } else {
    int mid = l + r >> 1;
    solve(l, mid, L);
    solve(mid + 1, r, R);
  }
  for (const Req &t : rqs[z]) {
    a[t.x] = 0;
  }
  diam = backup;
}

#undef L
#undef R

int main() {
  scanf("%d", &n);
  for (int x = 1; x <= n; ++x) {
    scanf("%d", a + x);
  }
  for (int i = 1; i < n; ++i) {
    int x, y;
    scanf("%d%d", &x, &y);
    e[x].eb(y), e[y].eb(x);
  }
  for (int k = 2; k <= n; ++k) {
    wg[k] = wg[k >> 1] + 1;
  }
  dfs(1);
  for (int k = 0; k < 17; ++k) {
    for (int i = 1; i + (1 << k + 1) - 1 < n; ++i) {
      f[k + 1][i] = sha(f[k][i], f[k][i + (1 << k)]);
    }
  }
  scanf("%d", &m);
  for (int i = 1; i <= m; ++i) {
    int x, c;
    scanf("%d%d", &x, &c);
    setrq(ts[x], i - 1, x, a[x]);
    ts[x] = i, a[x] = c;
  }
  for (int x = 1; x <= n; ++x) {
    setrq(ts[x], m, x, a[x]);
  }
  memset(a + 1, 0, n << 2);
  diam = (Diam){1, 1, 0};
  for (int z = 1; z <= n; ++z) {
    diam.upd(z);
  }
  solve();
  for (int i = 1; i <= m; ++i) {
    printf("%d\n", res[i] + 1 >> 1);
  }
  return 0;
}
```
